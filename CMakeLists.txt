cmake_minimum_required(VERSION 3.24)
project(Manhunt2Localization VERSION 0.1.0 LANGUAGES C CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Windows x64 with clang-cl or MSVC is required (GDI font rasterization).")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
include(cmake/Dependencies.cmake)

add_library(mh2loc_core STATIC
    src/json.cpp src/unicode.cpp src/file_io.cpp src/mh2_gxt.cpp
    src/mh2_dds.cpp src/mh2_tex.cpp src/mh2_font_dat.cpp
    src/mh2_font_build.cpp src/mh2_font_preview.cpp
    src/project.cpp src/package.cpp src/safe_files.cpp src/translations.cpp)
target_include_directories(mh2loc_core PUBLIC src)
target_compile_definitions(mh2loc_core PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_definitions(mh2loc_core PRIVATE MH2LOC_GENERATOR_VERSION="${PROJECT_VERSION}")
target_compile_options(mh2loc_core PRIVATE /W4 /permissive- /utf-8)
target_link_libraries(mh2loc_core PUBLIC zlibstatic Gdiplus Gdi32 Bcrypt)
add_executable(mh2loc src/main.cpp)
target_link_libraries(mh2loc PRIVATE mh2loc_core)
target_compile_options(mh2loc PRIVATE /W4 /permissive- /utf-8)
target_compile_definitions(mh2loc PRIVATE MH2LOC_VERSION="${PROJECT_VERSION}")

include(CTest)
if(BUILD_TESTING)
    add_executable(mh2loc_tests tests/tests.cpp)
    target_link_libraries(mh2loc_tests PRIVATE mh2loc_core)
    target_compile_options(mh2loc_tests PRIVATE /W4 /permissive- /utf-8)
    foreach(suite IN ITEMS config codecs builder deployment)
        add_test(NAME core.${suite} COMMAND mh2loc_tests ${suite} "${CMAKE_BINARY_DIR}/test-output")
        set_tests_properties(core.${suite} PROPERTIES TIMEOUT 180)
    endforeach()
    find_program(POWERSHELL_EXECUTABLE NAMES pwsh powershell REQUIRED)
    add_test(NAME cli.workflow COMMAND "${POWERSHELL_EXECUTABLE}" -NoProfile -ExecutionPolicy Bypass
        -File "${CMAKE_SOURCE_DIR}/tests/integration.ps1" -Executable "$<TARGET_FILE:mh2loc>"
        -Distribution "${CMAKE_SOURCE_DIR}/distribution" -TestRoot "${CMAKE_BINARY_DIR}/cli-test-output")
    set_tests_properties(cli.workflow PROPERTIES TIMEOUT 180)
endif()

install(TARGETS mh2loc RUNTIME DESTINATION .)
install(DIRECTORY distribution/ DESTINATION .)
install(FILES README.md README.zh-CN.md LICENSE NOTICE.md CHANGELOG.md DESTINATION .)
install(DIRECTORY docs/ DESTINATION docs)
install(FILES third_party/zlib-ng-LICENSE.md DESTINATION THIRD_PARTY)
set(CPACK_GENERATOR ZIP)
set(CPACK_PACKAGE_NAME mh2loc)
set(CPACK_PACKAGE_VERSION "${PROJECT_VERSION}")
set(CPACK_PACKAGE_FILE_NAME "mh2loc-${PROJECT_VERSION}-windows-x64")
set(CPACK_INCLUDE_TOPLEVEL_DIRECTORY ON)
set(CPACK_PACKAGE_CHECKSUM SHA256)
include(CPack)
